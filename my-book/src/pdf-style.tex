
% ============================================================
% Custom Pandoc PDF Styling
% No Eisvogel dependency
% ============================================================

% ------------------------------------------------------------
% Page geometry and book binding
% ------------------------------------------------------------

\usepackage[
    letterpaper,
    twoside,
    top=1in,
    bottom=1in,
    inner=1in,
    outer=1in,
    bindingoffset=0.25in,
    includehead,
    includefoot
]{geometry}

% ------------------------------------------------------------
% Colors
% ------------------------------------------------------------

\usepackage{xcolor}

\definecolor{codebackground}{HTML}{F0F0F0}
\definecolor{quotebackground}{HTML}{F5F5F5}
\definecolor{quoteborder}{HTML}{B8B8B8}

% ------------------------------------------------------------
% Code block formatting
% ------------------------------------------------------------

\usepackage{framed}
\usepackage{fvextra}

\definecolor{shadecolor}{HTML}{F0F0F0}

\renewenvironment{Shaded}
    {\def\FrameCommand{\fboxsep=8pt
     \colorbox{codebackground}}%
     \MakeFramed{\advance\hsize-\width
     \FrameRestore}}
    {\endMakeFramed}

% ------------------------------------------------------------
% Block quotation formatting
% ------------------------------------------------------------

\usepackage{mdframed}

\renewenvironment{quote}
    {\begin{mdframed}[
        backgroundcolor=quotebackground,
        linecolor=quoteborder,
        linewidth=2pt,
        topline=false,
        bottomline=false,
        rightline=false,
        leftline=true,
        innerleftmargin=12pt,
        innerrightmargin=10pt,
        innertopmargin=8pt,
        innerbottommargin=8pt,
        skipabove=10pt,
        skipbelow=10pt
    ]}
    {\end{mdframed}}

% ------------------------------------------------------------
% Typography
% ------------------------------------------------------------

\usepackage{microtype}

% ------------------------------------------------------------
% Paragraph spacing
% ------------------------------------------------------------

\setlength{\parskip}{6pt}
\setlength{\parindent}{0pt}

% ------------------------------------------------------------
% Header and footer
% ------------------------------------------------------------

\usepackage{fancyhdr}

\pagestyle{fancy}
\fancyhf{}

\fancyhead[LE,RO]{\thepage}

\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% ------------------------------------------------------------
% End custom styling
% ------------------------------------------------------------
